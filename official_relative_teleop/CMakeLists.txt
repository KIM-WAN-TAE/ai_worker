cmake_minimum_required(VERSION 3.16)
project(official_relative_teleop)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
# Preserve the installed Mini Leader's edge-trigger behaviour without changing upstream.
add_executable(gripper_trigger ../ffw_joint_trajectory_command_broadcaster/src/gripper_trigger.cpp)
target_include_directories(gripper_trigger PRIVATE ../ffw_joint_trajectory_command_broadcaster/include)
ament_target_dependencies(gripper_trigger rclcpp sensor_msgs std_msgs)
install(TARGETS gripper_trigger DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/compatibility_bridge.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config patches DESTINATION share/${PROJECT_NAME})
install(FILES upstream_versions.json README.md DESTINATION share/${PROJECT_NAME})
ament_package()
